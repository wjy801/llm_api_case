def test_quality_pipeline_preserves_the_stage_order(monkeypatch, tmp_path):
    events = []
    merge_result = SimpleNamespace(
        integrity_status=IntegrityStatus.COMPLETE,
        integrity_issues=(),
    )
    monkeypatch.setattr(
        quality_pipeline.quality_fact_merge_stage,
        "merge_quality_facts",
        lambda *args, **kwargs: events.append("fact-merge") or merge_result,
    )
    monkeypatch.setattr(
        quality_pipeline.quality_run_record,
        "write_final_run_record",
        lambda *args, **kwargs: events.append("run-record"),
    )
    monkeypatch.setattr(
        quality_pipeline.quality_semantic_stage,
        "run_semantic_stage",
        lambda *args, **kwargs: events.append("semantic"),
    )
    monkeypatch.setattr(
        quality_pipeline.quality_metrics_stage,
        "run_metrics_stage",
        lambda *args, **kwargs: events.append("metrics"),
    )
    imported = object()
    monkeypatch.setattr(
        quality_pipeline.quality_flaky_stage,
        "run_flaky_history_stage",
        lambda *args, **kwargs: events.append("flaky-import") or imported,
    )

    def state_stage(config, result):
        assert result is imported
        events.append("flaky-state")

    monkeypatch.setattr(
        quality_pipeline.quality_flaky_stage,
        "run_flaky_state_stage",
        state_stage,
    )
    quality_pipeline.finalize_quality_run(
        _config(tmp_path),
        start_time=datetime(2026, 8, 1, tzinfo=UTC),
        expected_execution_ids=("serial-pool",),
        expected_case_count=1,
        junit_files=(),
        status=RunStatus.FINISHED,
    )

    assert events == [
        "fact-merge",
        "run-record",
        "semantic",
        "metrics",
        "flaky-import",
        "flaky-state",
    ]

Runner把计划、pool终点和退出码原子写成执行Artifact

_write_execution_result()不依赖JUnit推测Runner状态，它直接记录计划nodeid和每个pool原始事实。

run_orchestration/runner.py::_write_execution_result · 完整函数
def _write_execution_result(
    *,
    test_path: str,
    argument_plan: pytest_execution.PytestArgumentPlan,
    collection: pytest_execution.CollectionResult,
    pool_results: tuple[pytest_execution.PoolExecutionResult, ...],
    final_exit_code: int,
) -> int:
    payload = {
        "schema_version": artifacts.RUNNER_EXECUTION_SCHEMA_VERSION,
        "test_target": str(test_path),
        "selection_args": list(argument_plan.selection_args),
        "planned_case_count": len(collection.cases),
        "planned_nodeids": [case.nodeid for case in collection.cases],
        "collection_exit_code": collection.raw_pytest_exit_code,
        "pool_results": [
            {
                "stage_id": result.stage_id,
                "planned_nodeids": list(result.planned_nodeids),
                "status": result.status.value,
                "raw_pytest_exit_code": result.raw_pytest_exit_code,
                "started_at": (
                    result.started_at.isoformat()
                    if result.started_at is not None
                    else None
                ),
                "completed_at": (
                    result.completed_at.isoformat()
                    if result.completed_at is not None
                    else None
                ),
                "exception_type": result.exception_type,
                "junit_path": (
                    result.junit_path.as_posix()
                    if result.junit_path is not None
                    else None
                ),
            }
            for result in pool_results
        ],
        "final_exit_code": final_exit_code,
    }
    try:
        artifacts.write_execution_result_atomic(payload)
    except Exception as error:
        print(
            f"Runner execution result write failed: "
            f"{type(error).__name__}: {error}",
            file=sys.stderr,
        )
        if final_exit_code in pytest_execution.PYTEST_TERMINATING_EXIT_CODES:
            return final_exit_code
        return pytest_execution.PYTEST_EXIT_TESTS_FAILED
    return final_exit_code

执行产物所有权

输入

collection冻结计划集合；pool_results保存stage_id、实际终点、时间和JUnit路径。

状态变化

payload通过write_execution_result_atomic()写入；成功后返回原final_exit_code。

同轮证据

Day30人工核对target、planned nodeids、serial-pool与P0 expected_execution_ids。

失败边界

执行Artifact写失败不能留下假绿：非终止退出会升级为tests failed。

3

finalize_quality_run()把P0提交变成后续消费者的可信起点

Merge失败立即停止；成功后先写最终run记录，再运行Semantic、Metrics和可选Flaky。

run_orchestration/quality_pipeline.py::finalize_quality_run · 完整函数
def finalize_quality_run(
    quality_config: QualityRuntimeConfig,
    *,
    start_time: datetime,
    expected_execution_ids: tuple[str, ...],
    expected_case_count: int,
    junit_files: tuple[Path | None, ...],
    status: RunStatus,
) -> None:
    if not quality_config.enabled or not quality_config.run_id:
        return
    merge_result = quality_fact_merge_stage.merge_quality_facts(
        quality_config,
        start_time=start_time,
        expected_execution_ids=expected_execution_ids,
        expected_case_count=expected_case_count,
        junit_files=junit_files,
    )
    if merge_result is None:
        return

    quality_run_record.write_final_run_record(
        quality_config,
        start_time=start_time,
        end_time=datetime.now(UTC),
        status=status,
        integrity_status=merge_result.integrity_status,
        integrity_issues=merge_result.integrity_issues,
    )
    quality_semantic_stage.run_semantic_stage(quality_config)
    quality_metrics_stage.run_metrics_stage(quality_config)
    flaky_import_result = quality_flaky_stage.run_flaky_history_stage(
        quality_config, status=status
    )
    quality_flaky_stage.run_flaky_state_stage(
        quality_config, flaky_import_result
    )

Quality事务边界

输入验证

Quality开关和run_id缺失时整个链NOOP；expected execution/count/JUnit进入P0完整性合并。

状态变化

def write_execution_result_atomic(
    payload: Mapping[str, object],
    path: str | Path = DEFAULT_EXECUTION_RESULT_PATH,
) -> Path:
    target = Path(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    temporary_path: Path | None = None
    try:
        with tempfile.NamedTemporaryFile(
            mode="w",
            encoding="utf-8",
            newline="\n",
            prefix=f".{target.name}.",
            suffix=".tmp",
            dir=target.parent,
            delete=False,
        ) as temporary_file:
            temporary_path = Path(temporary_file.name)
            json.dump(
                dict(payload),
                temporary_file,
                ensure_ascii=False,
                indent=2,
                sort_keys=True,
            )
            temporary_file.write("\n")
            temporary_file.flush()
            os.fsync(temporary_file.fileno())
        os.replace(temporary_path, target)
        temporary_path = None
        return target
    finally:
        if temporary_path is not None:
            temporary_path.unlink(missing_ok=True)

同轮与人工核对

代码自动证明

Runner Artifact原子替换；Reporting四种视图共享一个PipelineReport；Quality阶段顺序固定。

人工核对

脚本还要检查唯一run_id、serial-pool execution_id、target/planned nodeids、P0/Semantic/Metrics引用和mtime窗口。

产物所有权

Runner、JUnit、Allure、Quality与Pipeline各自产生文件；Day30证据目录只汇总，不改写内容。

失败边界

Markdown与JSON来自同一Report仍不等于脚本逐字段比对；答辩必须明确自动检查项与人工检查项。

01 · Question

“文件都存在”为什么仍不能证明同一轮

reports可能残留旧run、旧Manifest或旧execution-result；只看测试通过也无法证明Quality和Reporting消费了本轮事实。结业验收必须先隔离历史产物，再显式绑定唯一身份。

边界：不制造失败、不新增功能；Flaky默认关闭；真实Smoke禁止执行，Smoke只做collect-only；实际Runner目标固定为离线模块。

02 · Contract

脚本自动检查与人工答辩必须分栏

证据脚本实际自动检查仍需人工读取与答辩
Runner非NOT_RUN的pool stage_id为serial-pool，并与P0 expected_execution_ids一致。target、planned nodeids、execution-result中的final_exit_code和池级原始退出码。
Quality P0run_id一致、status=complete、expected_execution_ids一致。integrity_status、输出计数和完整性问题是否符合答辩结论。
Semantic/Metricsrun_id一致；Semantic status=complete；Metrics write_status=complete。Semantic p0_evidence、Metrics source_evidence及上游Hash引用是否一致。
ReportingPipeline JSON的质量观测阶段为PASSED，warnings没有run_id不一致。Markdown与JSON是否表达相同阶段、数量和结论；脚本没有逐字段比较两种视图。
时间与文件必需文件和Allure raw存在，核心文件mtime不早于本轮开始，JUnit数量可解析。文件内容是否完整支持Day29跨层答辩。
准确边界：内嵌PowerShell自检只自动覆盖中列列出的身份、提交状态、来源状态和时间事实；其余项目是答辩清单，不得宣称脚本已经验证。

03 · Flow

同轮验收是一条不可跳步的链

记录工作树→备份reports/allure→安全清理→唯一run_id→Quality全链开启→Smoke collect-only→离线Runner→更新阶段→生成MD+JSON→核对并留存→恢复原产物

04 · Cleanup

只清理两个经绝对路径校验的可再生产物目录

脚本把reports与allure-results解析为仓库内绝对路径，并同时校验目录叶名称；先复制到系统临时备份目录，再删除并重建。finally无论成功或失败都会恢复原目录和进程环境变量。

禁止：不得用通配符递归删除仓库根目录，不得先跨shell枚举再删除，不得在未备份时清空本地产物。

05 · Identity

父run_id显式设置，execution_id由Runner阶段派生

父身份

QUALITY_RUN_ID=day30-时间-Guid，所有Quality Manifest必须一致。

执行身份

串行Runner使用serial-pool，同时出现在execution-result与P0 expected_execution_ids。

能力开关

显式设置QUALITY、SEMANTIC和METRICS为TRUE。

Flaky

history/state均为FALSE，并移除进程级DB路径。

06 · Execution

Smoke只收集，实际执行目标是离线模块

先运行run_master.py module/smoke --collect-only并保存收集输出；随后用Runner执行module/offline_framework_example，输出smoke-tests.xml、allure-results、execution-result.json和完整Quality目录。

命名说明：Reporting合同把被验证的接口阶段命名为real_smoke；本地验收仅复用该阶段槽位，SMOKE_TARGET明确是离线模块，没有真实外部请求。

07 · Reporting

阶段状态先初始化，再按真实退出码更新

initialize-stages声明framework=false、collect=true、real-smoke=true；collect和离线Runner开始前为NO_DATA，结束后按退出码更新PASSED/FAILED。最后在同一环境中调用generate，同时写出Markdown与机器JSON。

initialize→set collect→set real_smoke→generate MD+generate JSON

08 · Verification

脚本自动核对身份、提交状态、来源状态和时间

  1. 四个Quality文件的run_id都等于本轮唯一值。
  2. P0 expected_execution_ids与Runner pool stage_id都为serial-pool。
  3. P0与Semantic status为complete，Metrics write_status为complete。
  4. Pipeline JSON中的质量观测阶段为PASSED，warnings不含run_id不一致。
  5. JUnit、execution-result、四个Manifest、Markdown与JSON均在本轮开始后写入。
脚本未自动检查：Runner target与planned nodeids、final_exit_code、P0 integrity_status、Semantic/Metrics上游引用，以及Markdown/JSON逐字段一致性；这些必须在答辩材料中人工核对。

Reporting的公开Report不重复暴露quality_run_id；它通过成功加载相同run.json与P0 Manifest后把质量阶段标记为PASSED，因此身份原件仍以Quality文件为准。

  [Environment]::SetEnvironmentVariable('PIPELINE_BUILD_RESULT', $BuildResult, 'Process')
  & $Python -m pipeline_reporting generate --workspace $RepoRoot --output reports/pipeline-summary.md --machine-output reports/pipeline-summary.json --dotenv .env
  if ($LASTEXITCODE -ne 0) { throw 'pipeline_reporting generate失败' }

  $RequiredFiles = @(
    'reports/smoke-tests.xml','reports/execution-result.json','reports/quality/run.json',
    'reports/quality/merged/manifest.json','reports/quality/semantic/merged/manifest.json',
    'reports/quality/metrics/manifest.json','reports/pipeline-summary.md','reports/pipeline-summary.json'
  ) | ForEach-Object { Join-Path $RepoRoot $_ }
  $Missing = @($RequiredFiles | Where-Object { !(Test-Path -LiteralPath $_ -PathType Leaf) })
  if ($Missing.Count) { throw ('缺少本轮产物：' + ($Missing -join ', ')) }
  if (!(Test-Path -LiteralPath $Allure -PathType Container) -or !(Get-ChildItem -LiteralPath $Allure -Recurse -File | Select-Object -First 1)) {
    throw 'Allure raw产物缺失'
  }
  $Run = Get-Content -LiteralPath (Join-Path $Quality 'run.json') -Raw -Encoding UTF8 | ConvertFrom-Json
  $P0 = Get-Content -LiteralPath (Join-Path $Quality 'merged\manifest.json') -Raw -Encoding UTF8 | ConvertFrom-Json
  $Semantic = Get-Content -LiteralPath (Join-Path $Quality 'semantic\merged\manifest.json') -Raw -Encoding UTF8 | ConvertFrom-Json
  $Metrics = Get-Content -LiteralPath (Join-Path $Quality 'metrics\manifest.json') -Raw -Encoding UTF8 | ConvertFrom-Json
  $Execution = Get-Content -LiteralPath (Join-Path $Reports 'execution-result.json') -Raw -Encoding UTF8 | ConvertFrom-Json
  $Report = Get-Content -LiteralPath (Join-Path $Reports 'pipeline-summary.json') -Raw -Encoding UTF8 | ConvertFrom-Json

  $Ids = @($Run.run_id, $P0.run_id, $Semantic.run_id, $Metrics.run_id)
  if (($Ids | Select-Object -Unique).Count -ne 1 -or $Ids[0] -ne $RunId) {
    throw ('Quality run_id不一致：' + ($Ids -join ', '))
  }
  $ExecutionIds = @($Execution.pool_results | Where-Object { $_.status -ne 'NOT_RUN' } | ForEach-Object { $_.stage_id })
  if (($ExecutionIds -join ',') -ne 'serial-pool' -or (@($P0.expected_execution_ids) -join ',') -ne 'serial-pool') {
    throw 'Runner与P0 execution_id不一致'
  }
  if ($P0.status -ne 'complete' -or $Semantic.status -ne 'complete' -or $Metrics.write_status -ne 'complete') {
    throw 'Quality Manifest尚未完整提交'
  }
  $QualityStage = @($Report.stages | Where-Object { $_.name -eq '质量观测' })
  if ($QualityStage.Count -ne 1 -or $QualityStage[0].status -ne 'PASSED') {
    throw 'Reporting未接受本轮Quality核心来源'
  }
  if (@($Report.warnings | Where-Object { $_ -match 'run_id.*不一致' }).Count) {
    throw 'Reporting存在run_id不一致警告'
  }
  foreach ($file in $RequiredFiles) {
    if ((Get-Item -LiteralPath $file).LastWriteTimeUtc -lt $StartedAtUtc.AddSeconds(-2)) {
      throw ('文件时间早于本轮：' + $file)
    }
  }

  [xml]$JUnit = Get-Content -LiteralPath (Join-Path $Reports 'smoke-tests.xml') -Raw -Encoding UTF8
  $Suites = @($JUnit.testsuites.testsuite)
  $Total = [int](($Suites | Measure-Object -Property tests -Sum).Sum)
  $Failed = [int](($Suites | Measure-Object -Property failures -Sum).Sum)
  $Errors = [int](($Suites | Measure-Object -Property errors -Sum).Sum)
  $Skipped = [int](($Suites | Measure-Object -Property skipped -Sum).Sum)
  $FinishedAtUtc = [DateTime]::UtcNow

  New-Item -ItemType Directory -Path $EvidenceDir -Force | Out-Null
  [ordered]@{
    run_id=$RunId; started_at=$StartedAtUtc.ToString('o'); finished_at=$FinishedAtUtc.ToString('o')
    collect_exit_code=$CollectExit; runner_exit_code=$RunnerExit; pipeline_conclusion=$Report.conclusion
    tests=$Total; passed=($Total-$Failed-$Errors-$Skipped); failed=$Failed; errors=$Errors; skipped=$Skipped
    worktree_before=$WorktreeBefore
  } | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath (Join-Path $EvidenceDir 'verification.json') -Encoding UTF8
  Copy-Item -LiteralPath $Reports -Destination (Join-Path $EvidenceDir 'reports') -Recurse
  Copy-Item -LiteralPath $Allure -Destination (Join-Path $EvidenceDir 'allure-results') -Recurse
  Write-Host ('同轮证据已保存：' + $EvidenceDir)
}
finally {
  Remove-WorkspaceArtifact $Reports 'reports'
  Remove-WorkspaceArtifact $Allure 'allure-results'
  if ($HadReports) { Copy-Item -LiteralPath (Join-Path $BackupRoot 'reports') -Destination $Reports -Recurse }
  if ($HadAllure) { Copy-Item -LiteralPath (Join-Path $BackupRoot 'allure-results') -Destination $Allure -Recurse }
  foreach ($name in $EnvNames) { [Environment]::SetEnvironmentVariable($name, $PreviousEnv[$name], 'Process') }
  Remove-Item -LiteralPath $BackupRoot -Recurse -Force -ErrorAction SilentlyContinue
}
Write-Host ('答辩使用的证据目录：' + $EvidenceDir)

12 · Design choice

设计选择与适用边界

现方案在一次连续、隔离的PowerShell会话中生成Runner、Quality与Reporting证据，并用身份、提交状态和时间证明同轮。
替代方案拼接此前成功运行的文件、截图和手工说明，或只以最终测试退出码作为结业证据。
选择理由同一会话减少环境漂移；唯一run_id、execution_id和写入时间构成可复查证据，已知失败也能被准确归属而非隐藏。
不适用条件生产发布验收、法规审计或跨组织交付通常还要求签名来源、独立审批、受控环境与长期留存，本地同轮证据不能替代这些保证。
